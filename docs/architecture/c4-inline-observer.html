sequenceDiagram
  participant Op as Operator
  participant API as /agents/{id}/run
  participant Bot as Demo Agent
  participant Tel as TelemetryService
  participant Det as Detector
  participant Orch as Orchestrator

  Op->>API: POST simulate
  API->>Bot: build_simulate()
  Bot-->>API: event + OTLP spans
  API->>Tel: ingest_otlp_json(promote=true)
  Tel->>Tel: parse + persist trajectory
  Tel->>Det: detect_anomaly(spans)
  Det-->>Tel: Detection(symptom)
  Tel->>Orch: process_event(AgentEvent)
  Orch-->>Tel: incident trace_id
  Tel-->>API: IngestResult
  API-->>Op: coagent + telemetry
